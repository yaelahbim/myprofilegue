MODUL 08 — INTEGRASI CLOUD

Konfigurasi Database Supabase & Skema SQL

Hubungkan proyek Supabase Anda agar seluruh perubahan di Admin Panel tersimpan online secara real-time dan dapat diakses dari perangkat mana pun.

1. Jalankan SQL Ini di Supabase SQL Editor

-- 1. Tabel utama penyimpan seluruh state portofolio
CREATE TABLE IF NOT EXISTS public.portfolio_state (
  id TEXT PRIMARY KEY DEFAULT 'main',
  content JSONB NOT NULL,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2. Tabel kotak masuk pesan pengunjung
CREATE TABLE IF NOT EXISTS public.portfolio_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  sender_name TEXT NOT NULL,
  sender_email TEXT NOT NULL,
  subject TEXT NOT NULL,
  message TEXT NOT NULL,
  is_read BOOLEAN NOT NULL DEFAULT FALSE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 3. Tabel analitik & trafik kunjungan
CREATE TABLE IF NOT EXISTS public.portfolio_visits (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  device_type TEXT NOT NULL,
  os_browser TEXT,
  city TEXT,
  region TEXT,
  country TEXT,
  country_code TEXT,
  timezone TEXT,
  screen_size TEXT,
  referrer TEXT,
  visited_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Jika tabel portfolio_visits sudah terlanjur dibuat sebelumnya, jalankan ALTER berikut:
ALTER TABLE public.portfolio_visits ADD COLUMN IF NOT EXISTS os_browser TEXT;
ALTER TABLE public.portfolio_visits ADD COLUMN IF NOT EXISTS city TEXT;
ALTER TABLE public.portfolio_visits ADD COLUMN IF NOT EXISTS region TEXT;
ALTER TABLE public.portfolio_visits ADD COLUMN IF NOT EXISTS country TEXT;
ALTER TABLE public.portfolio_visits ADD COLUMN IF NOT EXISTS country_code TEXT;
ALTER TABLE public.portfolio_visits ADD COLUMN IF NOT EXISTS timezone TEXT;

-- 4. Aktifkan Row Level Security (RLS)
ALTER TABLE public.portfolio_state ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.portfolio_messages ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.portfolio_visits ENABLE ROW LEVEL SECURITY;

-- 5. Kebijakan akses publik & admin terautentikasi
CREATE POLICY "Publik bisa membaca konten portofolio"
  ON public.portfolio_state FOR SELECT USING (true);

CREATE POLICY "Admin bisa menyimpan konten portofolio"
  ON public.portfolio_state FOR ALL
  USING (auth.role() = 'authenticated')
  WITH CHECK (auth.role() = 'authenticated');

CREATE POLICY "Publik bisa mengirim pesan kontak"
  ON public.portfolio_messages FOR INSERT WITH CHECK (true);

CREATE POLICY "Admin bisa kelola pesan kontak"
  ON public.portfolio_messages FOR ALL
  USING (auth.role() = 'authenticated');

CREATE POLICY "Publik bisa mencatat kunjungan"
  ON public.portfolio_visits FOR INSERT WITH CHECK (true);

CREATE POLICY "Admin bisa membaca trafik kunjungan"
  ON public.portfolio_visits FOR SELECT
  USING (auth.role() = 'authenticated');
